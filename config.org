** Use Package

Install packages by default when configured with use-package.

#+begin_src emacs-lisp
(setq use-package-always-ensure t)
#+end_src

Macro to use to configure built-in features, rather than packages that do not
need to be installed.

#+begin_src emacs-lisp
(defmacro use-feature (name &rest args)
  "`use-package' with package installation (`ensure') disabled."
  (declare (indent defun))
  `(use-package ,name
     :ensure nil
     ,@args))
#+end_src

#+end_src

** Customization Saving

Do not save customizations in =init.el=, instead save them in a temporary file
that will be discarded. As a result, any customizations made in a session will
not be persisted across sessions.

#+begin_src emacs-lisp
(setq custom-file (expand-file-name
                   (format "custom-%d-%d.el" (emacs-pid) (random))
                   temporary-file-directory))
#+end_src

** Project Tabs (Tabspaces)

Create a tab for each project, with buffers isolated per tab.

#+begin_src emacs-lisp
(use-package tabspaces
  :after projectile
  :hook (elpaca-after-init . tabspaces-mode)
  :bind (("C-c p t" . tabspaces-open-or-create-project-and-workspace)
         ("C-c p n" . my/open-neuron-project-tab)
         ("C-c p d" . my/open-project-tab-from-dir))
  :custom
  (tabspaces-use-filtered-buffers-as-default t)
  (tabspaces-default-tab "Default")
  (tabspaces-include-buffers '("*scratch*" "*Messages*"))
  :config
  (defvar my/neuron-projects-dir "~/workspace/wsgr/neuron/"
    "Base directory for neuron projects.")

  (defun my/open-project-in-tab (project-dir)
    "Open PROJECT-DIR in a new tab without prompts."
    (let* ((project-dir (expand-file-name project-dir))
           (tab-name (file-name-nondirectory (directory-file-name project-dir))))
      (unless (member project-dir projectile-known-projects)
        (projectile-add-known-project project-dir))
      (tab-bar-new-tab)
      (tab-bar-rename-tab tab-name)
      (setq default-directory project-dir)
      (dired project-dir)
      (when (featurep 'treemacs)
        (run-at-time 0 nil
          (lambda (dir name)
            (let ((ws (treemacs-current-workspace)))
              (when ws
                (setf (treemacs-workspace->projects ws)
                      (list (treemacs-project->create!
                             :name name
                             :path dir
                             :path-status (treemacs--get-path-status dir))))
                (when (treemacs-get-local-buffer)
                  (treemacs--invalidate-buffer-project-cache)
                  (treemacs--rerender-after-workspace-change)))))
          project-dir tab-name))))

  (defun my/open-neuron-project-tab ()
    "Open a neuron project in a new tab."
    (interactive)
    (let* ((dir (expand-file-name my/neuron-projects-dir))
           (subdirs (seq-filter
                     #'file-directory-p
                     (directory-files dir t "^[^.]")))
           (names (mapcar #'file-name-nondirectory subdirs))
           (choice (completing-read "Neuron project: " names nil t)))
      (my/open-project-in-tab (expand-file-name choice dir))))

  (defun my/open-project-tab-from-dir (dir)
    "Open a project tab from subdirectories of DIR."
    (interactive "DBase directory: ")
    (let* ((subdirs (seq-filter
                     #'file-directory-p
                     (directory-files dir t "^[^.]")))
           (names (mapcar #'file-name-nondirectory subdirs))
           (choice (completing-read "Project: " names nil t)))
      (my/open-project-in-tab (expand-file-name choice dir)))))

(use-feature tab-bar

  (tab-bar-new-tab-choice "*scratch*"))
#+end_src

* AI

** Claude Code IDE

Integrates Claude Code CLI directly into Emacs via MCP protocol. Uses your
existing Claude Code login — no API key needed.

- =C-c C-'= opens the command menu
- Automatic project detection with per-project sessions
- Ediff integration for reviewing code changes

#+begin_src emacs-lisp
(defun my/claude-dir-has-auth-p (claude-dir)
  "Return non-nil if CLAUDE-DIR has authentication credentials."
  (let ((config-file (expand-file-name ".claude.json" claude-dir)))
    (and (file-exists-p config-file)
         (with-temp-buffer
           (insert-file-contents config-file)
           (search-forward "oauthAccount" nil t)))))

(defun my/find-claude-config-dir ()
  "Search up from `default-directory' for .claude with auth, fall back to ~/.claude-sabrams."
  (let ((dir default-directory)
        (found nil))
    (while (and dir (not found))
      (let ((claude-dir (expand-file-name ".claude" dir)))
        (when (and (file-directory-p claude-dir)
                   (my/claude-dir-has-auth-p claude-dir))
          (setq found claude-dir)))
      (let ((parent (file-name-directory (directory-file-name dir))))
        (setq dir (unless (equal parent dir) parent))))
    (or found (expand-file-name "~/.claude-sabrams"))))

(defvar my/claude-current-profile nil
  "The Claude profile being used for the current session.")

(defun my/set-claude-config-dir (&rest _)
  "Set CLAUDE_CONFIG_DIR in vterm-environment before starting Claude."
  (let ((config-dir (my/find-claude-config-dir)))
    (setq my/claude-current-profile config-dir)
    (message "Using Claude profile: %s" config-dir)
    (setq vterm-environment
          (cons (concat "CLAUDE_CONFIG_DIR=" config-dir)
                (seq-remove (lambda (s) (string-prefix-p "CLAUDE_CONFIG_DIR=" s))
                            vterm-environment)))))

(defun my/claude-show-profile-in-buffer (&rest _)
  "Display the Claude profile in the buffer header line."
  (when my/claude-current-profile
    (run-at-time 0.5 nil
                 (lambda ()
                   (when-let ((buf (seq-find (lambda (b)
                                               (string-prefix-p "*claude-code" (buffer-name b)))
                                             (buffer-list))))
                     (with-current-buffer buf
                       (setq-local header-line-format
                                   (format " Profile: %s" my/claude-current-profile))))))))

(use-package claude-code-ide
